33// RAII wrapper around IpplTimings::startTimer / stopTimer so per-function
34// instrumentation in the manager pipeline is exception-safe and clutter-free.
35// Lifetime: starts on construction, stops on destruction. Use with a
36// static-local TimerRef so the timer is registered once per call site:
37//
38// static IpplTimings::TimerRef t = IpplTimings::getTimer("kickHalf");
39// ScopedIpplTimer scope(t);
40class ScopedIpplTimer {
41public:
42 explicit ScopedIpplTimer(IpplTimings::TimerRef ref) : ref_(ref) {
43 IpplTimings::startTimer(ref_);
44 }
45 ~ScopedIpplTimer() { IpplTimings::stopTimer(ref_); }
46 ScopedIpplTimer(const ScopedIpplTimer&) = delete;
47 ScopedIpplTimer& operator=(const ScopedIpplTimer&) = delete;
48
49private:
50 IpplTimings::TimerRef ref_;
51};

53// Base class for gridless Barnes-Hut simulations driven by IPPL's BaseManager
54// loop (pre_step / advance / post_step). Mirrors the role AlpineManager plays
55// for the FFT-PIC stack: owns the particle container and the BH solver, exposes
56// the physics-agnostic helpers (kickHalf, drift, wrap, solve), and delegates
57// the simulation-specific pieces (IC sampling, solver parameters, per-step
58// ordering, diagnostics) to derived managers via pure-virtual hooks.
59//
60// Templated on a precision policy P (NBody/BHPrecision.hpp). All inner types
61// (container, solver, leapfrog, diagnostics) take the same P so a single
62// template parameter selects the precision config end-to-end. Available
63// policies are: DoublePrecision, MixedPrecision, FloatPrecision
64//
65// Lifetime / ownership:
66// - The base owns the container and the solver as unique_ptrs. Both are
67// constructed lazily inside pre_run(), because derived classes need to
68// supply construction parameters (box, BCs, theta, G, ...) that depend
69// on simulation-specific state.
70// - run(int nt) is inherited verbatim from ippl::BaseManager.
71//
72// Required overrides in derived classes:
73// initializeContainer() - construct pcontainer_m with the right box/BCs
74// initializeParticles() - fill positions, velocities, charges, h
75// advanceImpl() - one timestep
76// prepareSolverInputs() - syncGrav + exchangeHalos
77//
78// Optional overrides:
79// initializeSolverParams(Params&) - define G, theta, Ewald shells
80// post_initial_solve() - hook between t=0 BH solve and t=0 dump
81// (e.g. DIH calibrates focusing here)
82// dumpImpl() - write CSV row / stdout report
83//
84// Note the `Impl` suffix on advance / dump: the base class wraps both with a
85// ScopedIpplTimer so derived classes don't need to remember to instrument.
86template <class P, unsigned Dim>
87class NBodyManager : public ippl::BaseManager {
88 static_assert(Dim == 3, "NBodyManager requires Dim == 3");
89
90public:
91 using Precision = P;
92 using Tc = typename P::Tc;
93 using Container = NBodyParticleContainer<P, Dim>;
94 using Solver = NBodySolver<P, Dim>;
95 using SolverParams = typename Solver::Params;

108 // BaseManager interface --------------------------------------------------

110 // Standard sequence: build container → allocate → fill IC → build solver
111 // → first BH evaluation → optional post-solve hook → dump t=0 row. Derived
112 // classes that need a different sequence can override pre_run() outright.
113 void pre_run() override {
114 static IpplTimings::TimerRef tInit = IpplTimings::getTimer("init");
115 static IpplTimings::TimerRef tFirstSolv = IpplTimings::getTimer("firstSolve");
116
117 {
118 ScopedIpplTimer scope(tInit);
119 initializeContainer();
120 // Allocate per-rank storage. Derived managers fill only the local
121 // slice [0, localN()); after the first sync (syncGravBH) inside
122 // prepareSolverInputs(), cstone reshards globally and [startIndex(),
123 // endIndex()) becomes this rank's owned range.
124 // Halo particles are stored in [0,startIndex()) and [endIndex(), N).
125 const unsigned n0 = static_cast<unsigned>(localN());
126 pcontainer_m->create(n0);
127 initializeParticles();
128
129 SolverParams bhParams;
130 initializeSolverParams(bhParams);
131 solver_m = std::make_unique<Solver>(*pcontainer_m, bhParams);
132 }
133
134 {
135 ScopedIpplTimer scope(tFirstSolv);
136 // warmup=true: the t=0 solve is timed only by the outer
137 // "firstSolve" timer; the per-scope bh.* accumulators and the
138 prepareSolverInputs(/*collect=*/false);
139 solver_m->runSolver(/*warmup=*/true);
140 }
141 post_initial_solve();
142
143 time_m = Tc(0);
144 it_m = 0;
145 dump();
146 }
147
148 // BaseManager::advance() is pure-virtual. NBodyManager::advance() wraps
149 // advanceImpl() in a "step" timer. Derived classes override advanceImpl().
150 void advance() final {
151 static IpplTimings::TimerRef t = IpplTimings::getTimer("step");
152 ScopedIpplTimer scope(t);
153 advanceImpl();
154 }
155
156 // Default post_step: advance time and emit a CSV row. Derived classes
157 // override dumpImpl() rather than post_step() unless they need a different
158 // ordering of time/it/dump.
159 void post_step() override {
160 time_m += dt_m;
161 ++it_m;
162 dump();
163 }
164
165 int getNt() const { return Nt_m; }

167protected:

185 }

227 // Per-rank slice [firstGlobal, lastGlobal) of the global N. Even split with
228 // remainder on the trailing ranks: rank r owns [r*N/R, (r+1)*N/R). Single
229 // rank: [0, N) so existing single-rank behavior is unchanged.
230 unsigned long firstGlobal() const {
231 return (N_m * static_cast<unsigned long>(rank())) /
232 static_cast<unsigned long>(numRanks());
233 }
234 unsigned long lastGlobal() const {
235 return (N_m * static_cast<unsigned long>(rank() + 1)) /
236 static_cast<unsigned long>(numRanks());
237 }
238 unsigned long localN() const { return lastGlobal() - firstGlobal(); }

242 pcontainer_m = std::move(pc);
243 }

253 int it_m{0};
254};
