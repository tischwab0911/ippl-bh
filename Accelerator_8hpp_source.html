28// Build-global execution policy, chosen at configure time exactly like sphexa's
29// USE_CUDA-driven AccType. The vendored cstone_gpu target defines USE_CUDA
30// PUBLIC, so any GPU build (CUDA or HIP) sees it; the CPU build links only
31// cstone_headers and never defines it. kExec is the policy object cstone's
32// primitives take as first argument: the default GPU stream, or the CPU tag.
33#if defined(USE_CUDA)
34using NBodyAcc = cstone::execution::Gpu;
35inline constexpr NBodyAcc kExec = cstone::execution::gpuDefaultStream;
36#else
37using NBodyAcc = cstone::execution::Cpu;
38inline constexpr NBodyAcc kExec = cstone::execution::cpu;
39#endif
40
41// This inline expression is used to select between CPU and GPU implementations
42inline constexpr bool kHaveGpu = bool(cstone::execution::HaveGpu<NBodyAcc>{});
43
44// Block until all work queued on kExec has finished; a no-op on the CPU build,
45// whose cstone/ryoanji kernels are synchronous. (cstone::syncGpu only accepts a
46// GPU stream, so it cannot sit in a discarded if-constexpr branch of a CPU build.)
47inline void syncExec() {
48#if defined(USE_CUDA)
49 cstone::syncGpu(kExec);
50#endif
51}

53// Per-component field storage: GPU-resident DeviceVector on device builds, plain
54// std::vector on the CPU build. Mirrors sphexa ParticlesData::FieldVector.
55template <class T>
56using FieldVector =
57 std::conditional_t<kHaveGpu, cstone::DeviceVector<T>, std::vector<T>>;
