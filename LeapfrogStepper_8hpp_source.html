26// KDK leapfrog primitives over the locally-owned range [startIndex(), endIndex()).
27// Velocity (Px/Py/Pz) and the field (Ex/Ey/Ez) are container-resident. The drift
28// folds the periodic wrap (cstone::putInBox) into the position update, the way
29// sphexa's positionUpdate does. Each primitive dispatches at compile time via
30// kHaveGpu to a GPU kernel (helpers/primitives_gpu.cu) or the inline OpenMP loops
31// below.
32
33namespace detail {

35// CPU (OpenMP) bodies.
36template <class P>
37void leapfrogKickHalfCpu(NBodyParticleContainer<P, 3>& pc, typename P::Tc dt) {
38 using Tc = typename P::Tc;
39 using Ta = typename P::Ta;
40 const unsigned start = pc.startIndex();
41 const unsigned end = pc.endIndex();
42 if (end <= start) { return; }
43 const Ta* Ex = getRaw<"Ex">(pc);
44 const Ta* Ey = getRaw<"Ey">(pc);
45 const Ta* Ez = getRaw<"Ez">(pc);
46 Tc* px = getRaw<"Px">(pc); Tc* py = getRaw<"Py">(pc); Tc* pz = getRaw<"Pz">(pc);
47 const Tc halfDt = Tc(0.5) * dt;
48#pragma omp parallel for schedule(static)
49 for (unsigned i = start; i < end; ++i) {
50 px[i] -= halfDt * static_cast<Tc>(Ex[i]);
51 py[i] -= halfDt * static_cast<Tc>(Ey[i]);
52 pz[i] -= halfDt * static_cast<Tc>(Ez[i]);
53 }
54}

75template <class P>
76void leapfrogDriftCpu(NBodyParticleContainer<P, 3>& pc, typename P::Tc dt) {
77 using Tc = typename P::Tc;
78 const unsigned start = pc.startIndex();
79 const unsigned end = pc.endIndex();
80 if (end <= start) { return; }
81 const auto box = pc.box();
82 Tc* Rx = getRaw<"Rx">(pc);
83 Tc* Ry = getRaw<"Ry">(pc);
84 Tc* Rz = getRaw<"Rz">(pc);
85 const Tc* px = getRaw<"Px">(pc); const Tc* py = getRaw<"Py">(pc); const Tc* pz = getRaw<"Pz">(pc);
86#pragma omp parallel for schedule(static)
87 for (unsigned i = start; i < end; ++i) {
88 cstone::Vec3<Tc> X{Rx[i] + dt * px[i], Ry[i] + dt * py[i], Rz[i] + dt * pz[i]};
89 X = cstone::putInBox(X, box);
90 Rx[i] = X[0];
91 Ry[i] = X[1];
92 Rz[i] = X[2];
93 }
94}

98// Half-kick: P_i -= 0.5 * dt * E_i (axpy with s = -0.5*dt).
99template <class P>
100void leapfrogKickHalf(NBodyParticleContainer<P, 3>& pc, typename P::Tc dt) {
101 using Tc = typename P::Tc;
102 using Ta = typename P::Ta;
103 if constexpr (kHaveGpu) {
104 const unsigned start = pc.startIndex();
105 const unsigned n = pc.endIndex() - start;
106 detail::axpy3Gpu<Tc, Ta>(start, n,
107 getRaw<"Px">(pc), getRaw<"Py">(pc), getRaw<"Pz">(pc),
108 getRaw<"Ex">(pc), getRaw<"Ey">(pc), getRaw<"Ez">(pc),
109 -Tc(0.5) * dt);
110 } else {
111 detail::leapfrogKickHalfCpu(pc, dt);
112 }
113}

132// Drift + periodic wrap: R_i = putInBox(R_i + dt * P_i, box). Open-BC axes pass
133// through unchanged. No standalone wrap pass is needed afterwards.
134template <class P>
135void leapfrogDrift(NBodyParticleContainer<P, 3>& pc, typename P::Tc dt) {
136 using Tc = typename P::Tc;
137 if constexpr (kHaveGpu) {
138 const unsigned start = pc.startIndex();
139 const unsigned n = pc.endIndex() - start;
140 detail::driftWrapGpu<Tc>(start, n,
141 getRaw<"Rx">(pc), getRaw<"Ry">(pc), getRaw<"Rz">(pc),
142 getRaw<"Px">(pc), getRaw<"Py">(pc), getRaw<"Pz">(pc),
143 dt, pc.box());
144 } else {
145 detail::leapfrogDriftCpu(pc, dt);
146 }
