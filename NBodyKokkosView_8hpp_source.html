25// Consumer-side adapter: unmanaged Kokkos::View accessors over the container's
26// raw field storage. Included ONLY by Kokkos consumers (the .cu kernel TUs,
27// diagnostics, and the OPAL-X kicker) — never by the producer path. Keeping this
28// out of NBodyParticleContainer.hpp is what lets the g++ sync TU include the
29// container without dragging in Kokkos (and the -DKOKKOS_DEPENDENCE that would
30// reroute it through nvcc_wrapper).
31
32namespace ippl::nbody {
33
34// Memory space of the consumer-facing view: the device space IPPL configured
35// (CudaSpace/HIPSpace) on GPU builds, HostSpace on CPU — matching where the
36// FieldVector storage lives.
37using ViewSpace = std::conditional_t<kHaveGpu,
38 Kokkos::DefaultExecutionSpace::memory_space,
39 Kokkos::HostSpace>;
40template <class U>
41using KView = Kokkos::View<U*, ViewSpace, Kokkos::MemoryTraits<Kokkos::Unmanaged>>;
42template <class U>
43using KViewConst = Kokkos::View<const U*, ViewSpace, Kokkos::MemoryTraits<Kokkos::Unmanaged>>;
44
45// Unmanaged Kokkos::View over the storage of field Name.
46template <util::StructuralString Name, class P, unsigned Dim>
47auto getView(NBodyParticleContainer<P, Dim>& pc) {
48 auto& dv = pc.template getDV<Name>();
49 using T = typename std::remove_reference_t<decltype(dv)>::value_type;
50 return KView<T>(dv.data(), dv.size());
51}
52template <util::StructuralString Name, class P, unsigned Dim>
53auto getView(const NBodyParticleContainer<P, Dim>& pc) {
54 auto const& dv = pc.template getDV<Name>();
55 using T = typename std::remove_reference_t<decltype(dv)>::value_type;
56 return KViewConst<T>(dv.data(), dv.size());
57}
