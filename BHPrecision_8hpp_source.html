20// Precision policy for the gridless Barnes-Hut stack. Each policy bundles the
21// five scalar template parameters that ryoanji's traversal kernel exposes,
22// plus the multipole value type. The three policies below correspond one-to-one
23// to ryoanji's pre-compiled TRAVERSE_MPOLE instantiations in
24// extern/ryoanji/src/ryoanji/nbody/traversal_gpu.cu — adding a fourth combo
25// here without a matching TRAVERSE row in ryoanji will produce a link error.
26//
27// Type roles:
28// Tc — coordinate type for positions (R), velocity (P), box, cstone domain
29// Th — smoothing length (h) as consumed by ryoanji P2P
30// Tm — mass / charge as consumed by ryoanji P2P
31// Ta — accelerations (Ex/Ey/Ez) as written by ryoanji and consumed by leapfrog
32// Tf — multipole expansion-center coordinate type (Vec4<Tf> source centers)
33// Tmm — multipole *value* type (template arg to CartesianQuadrupole<...>)
34
35struct DoublePrecision {
36 using Tc = double;
37 using Th = double;
38 using Tm = double;
39 using Ta = double;
40 using Tf = double;
41 using Tmm = double;
42};
