41// Per-axis sum + local count -> global mean. Shared by both backends.
42template <class T>
43inline Triple<T> finalizeMeanTriple(Triple<T> localSum, unsigned long localN, MPI_Comm comm) {
44 T inout[4] = {localSum.x, localSum.y, localSum.z, static_cast<T>(localN)};
45 MPI_Allreduce(MPI_IN_PLACE, inout, 4, ::MpiType<T>{}, MPI_SUM, comm);
46 const T invN = (inout[3] > T(0)) ? T(1) / inout[3] : T(0);
47 return Triple<T>{inout[0] * invN, inout[1] * invN, inout[2] * invN};
48}

102void shiftRight(T* data, unsigned first, unsigned last, unsigned by) {

104 if constexpr (kHaveGpu) { shiftRightGpu(data, first, last, by); }
