51// Each container kernel is a backend dispatcher: the GPU body is a kernel launch
52// defined in NBodyParticleContainer.cu (declared here, explicitly instantiated
53// there), the CPU body is the inline OpenMP loop in the if constexpr(!kHaveGpu)
54// branch. Mirrors the LeapfrogStepper split, so the container needs only a .cu —
55// no per-backend .cpp.
56
57// Geometric-mean edge length of a level-`level` octree cell, cbrtVol = cbrt(box volume).
58// reductions[d] = maxTreeLevel - (SFC key bits of axis d) under cstone's mixed-dimension
59// keys: axis d is not split above level reductions[d], so its cell edge is
60// l_d * 2^(min(reductions[d], level) - level). For a cubic box all reductions are 0 and the
61// edge is cbrtVol / 2^level.
62template <class Th>
63HOST_DEVICE_FUN inline Th cellEdge(Th cbrtVol, unsigned level, cstone::AxesBits reductions) {
64 unsigned unsplit = 0;
65 for (int d = 0; d < 3; ++d) {
66 unsplit += reductions[d] < level ? reductions[d] : level;
67 }
68 return cbrtVol * Th(exp2(double(unsplit) / 3.0 - double(level)));
69}

71// Per-particle leaf-derived smoothing length: h[i] = cellEdge(level(leaf(i))).
72// Both backends call the same host/device cstone helpers (findNodeBelow,
73// treeLevel). Instantiated for (Th, KeyType).
74template <class Th, class KeyType>
75void setHFromLeavesGpu(const KeyType* keys, const KeyType* leaves, int nLeafKeys,
76 Th cbrtVol, cstone::AxesBits reductions, Th* h, unsigned n);
77
78template <class Th, class KeyType>
79void setHFromLeaves(const KeyType* keys, const KeyType* leaves, int nLeafKeys,
80 Th cbrtVol, cstone::AxesBits reductions, Th* h, unsigned n) {
81 if constexpr (kHaveGpu) {
82 setHFromLeavesGpu(keys, leaves, nLeafKeys, cbrtVol, reductions, h, n);
83 } else {
84#pragma omp parallel for schedule(static)
85 for (unsigned i = 0; i < n; ++i) {
86 KeyType k = keys[i];
87 cstone::TreeNodeIndex leafIdx = cstone::findNodeBelow(leaves, nLeafKeys, k);
88 KeyType codeRange = leaves[leafIdx + 1] - leaves[leafIdx];
89 unsigned level = cstone::treeLevel(codeRange);
90 h[i] = cellEdge(cbrtVol, level, reductions);
91 }
92 }
93}

132// sphexa-style conserved/dependent sync entry points:
133//
134// syncGravBH<P, ConservedFields, DependentFields>(pc)
135// cstone Domain::syncGrav: SFC-sort + redistribute + halo discovery,
136// threading ConservedFields in lockstep with positions, resizing
137// DependentFields to nParticlesWithHalos, then refilling h for ryoanji P2P
138// softening. The multipole-mass slot is idxOf<"charge">.
139//
140// updateBH<P, ConservedFields>(pc)
141// non-gravity Domain::sync (used by tests).
142//
143// ConservedFields / DependentFields are cstone util::FieldList<...> selections
144// over fieldNames (see fields::StdConserved / fields::StdDependent). Each must
145// list >= 2 fields — cstone's get<FieldList> collapses a single-field list to a
146// bare reference, which the tuple-based sync API does not accept.
147//
148// Defined in NBodySync.cpp — a Kokkos-free translation unit compiled by plain
149// g++, the only place cstone::get<FieldList<...>> (C++20 NTTP-pack CTAD that nvcc
150// rejects) is instantiated. Caller TUs (compiled via nvcc_wrapper) see only these
151// declarations and link the g++ instantiations, so the forbidden CTAD never
152// reaches nvcc; every (P, ConservedFields, DependentFields) combo a caller uses
153// must be explicitly instantiated in NBodySync.cpp. Declared here as friends so
154// they can drive the private sort scratch and the post-sync h refill.
155template <class P, class ConservedFields, class DependentFields>
156void syncGravBH(NBodyParticleContainer<P, 3>& pc);
157template <class P, class ConservedFields>
158void updateBH(NBodyParticleContainer<P, 3>& pc);

160// Gridless particle container wrapping cstone::Domain. Doubles as the sphexa
161// "DataType": its public field members (x, y, z, h, m, ax, ay, az, ugrav) and
162// the AcceleratorType / KeyType / RealType typedefs are exactly the surface the
163// vendored gravity_wrapper.hpp MultipoleHolder{Cpu,Gpu} read, while fieldNames +
164// dataTuple() are the surface cstone::get<FieldList<...>> reads. The friendly
165// getRaw<"Name">/idxOf/getDV accessors map the legacy names onto these members
166// (Rx->x, charge->m, Ex->ax, ...), so existing call sites are unchanged.
167//
168// Kokkos-free by design: consumer-side unmanaged-View accessors live in
169// NBody/NBodyKokkosView.hpp, included only by Kokkos/.cu consumers. This header
170// stays raw-pointer / cstone only, so the g++ sync TU that includes it links
171// against cstone_gpu without dragging in Kokkos.
172//
173// Field storage is FieldVector<T>: cstone::DeviceVector<T> on GPU builds, plain
174// std::vector<T> on the CPU build. The accelerator (NBodyAcc) is build-global,
175// selected by USE_CUDA — one AccType per build, like sphexa.

254 // Replace the cstone domain by a fresh one (new bucket sizes and box) and keep every buffer
255 // of the container with its capacity. All particles and halos are dropped: the container is
256 // empty as after construction, and the next create() and sync start the new domain from
257 // scratch. Rebuilding the container instead frees buffers that GPU-aware MPI has used, and
258 // Cray MPICH's GPU IPC keeps freed device buffers that a peer mapped: device memory grew by
259 // the size of those buffers at every rebuild.
260 void resetDomain(unsigned bucketSize, unsigned bucketSizeFocus, float theta,
261 const cstone::Box<Tc>& box) {
262 int rank = 0;
263 MPI_Comm_rank(comm_, &rank);
264 // Construct in place: assigning a temporary copies it (cstone's DeviceVector has no move),
265 // and views the domain caches into its own buffers would keep pointing into the
266 // destroyed temporary.
267 std::destroy_at(&domain_);
268 std::construct_at(&domain_, kExec, rank, static_cast<int>(nRanks_m), bucketSize,
269 bucketSizeFocus, theta, comm_, box);
270 // Empty arrays (capacity kept): the first sync of the new domain takes the array size as
271 // its particle count, so a rank that packs nothing (getLocalNum() == 0 == n, no create())
272 // would otherwise hand the old domain's particles to it.
273 util::for_each_tuple([](auto& a) { a.resize(0); }, dataTuple());
274 keys.resize(0);
275 hZero.resize(0);
276 }

283 // Inject nLocal new locally-owned particles (mirrors ParticleBase::create —
284 // incremental delta + auto unique IDs). Grows every registry field + keys +
285 // the dummy-h buffer to nParticlesWithHalos()+nLocal, preserving existing data
286 // (reallocate reserves only when capacity is short), opens the gap
287 // [end, end+nLocal) by shifting the suffix halos right, fills valid keys and
288 // unique strided IDs for the new slots, and extends the assigned range. The
289 // next syncGravBH redistributes the new particles to their owner ranks and
290 // refills halos. Returns the offset; the caller writes positions / velocity /
291 // charge / h at [offset, offset+nLocal) via getRaw before that sync.
292 //
293 // The init call (empty domain) reproduces a plain nLocal allocation.
294 LocalIndex create(LocalIndex nLocal) {
295 const LocalIndex end = domain_.endIndex();
296 const LocalIndex size = domain_.nParticlesWithHalos();
297 const LocalIndex newSize = size + nLocal;
298
299 auto t = dataTuple();
300 util::for_each_tuple([this, newSize](auto& a) { reallocate(a, newSize, allocGrowthRate_); }, t);
301 reallocate(keys, newSize, allocGrowthRate_);
302 reallocate(hZero, newSize, allocGrowthRate_);
303 cstone::fill(kExec, hZero.data(), hZero.data() + newSize, Th(0));
304 for (auto* a : userAttribs) a->resize(newSize);
305
306 // Open the gap for the new particles: relocate the suffix-halo payload
307 // [end, size) -> [end+nLocal, newSize). Overlap-safe (see detail::shiftRight).
308 // Keys are not relocated — cstone recomputes them from positions on the next
309 // sync, reading only the removeKey sentinel that the fill below clears.
310 util::for_each_tuple([end, size, nLocal](auto& a) { detail::shiftRight(a.data(), end, size, nLocal); }, t);
311
312 // New particles occupy [end, end+nLocal): valid (non-removeKey) keys so the
313 // next sync includes them; unique IDs strided by rank (like ParticleBase).
314 // Zero the whole suffix, not only the new slots: the shifted halos' key slots
315 // hold stale values and the grown tail is uninitialized memory. A stale removeKey
316 // there would be inherited by a particle that the next sync receives into it.
317 cstone::fill(kExec, keys.data() + end, keys.data() + newSize, KeyType(0));
318 detail::assignIds(id.data(), end, nLocal, nextId_m, nRanks_m);
319 nextId_m += nRanks_m * static_cast<IdType>(nLocal);
320
321 // Advertise the grown buffer (assigned range end + the shifted suffix halos)
322 // so the next sync sizes its exchange/scratch to newSize. setEndIndex alone
323 // leaves nParticlesWithHalos() stale, which only self-corrects on the domain's
324 // first sync (initBounds); a later create() would otherwise under-size the
325 // exchange buffers and corrupt the heap during the gather.
326 domain_.setEndIndex(end + nLocal);
327 domain_.setNParticlesWithHalos(newSize);
328 return end;
329 }

331 // Flag locally-owned particles for removal at the next sync (mirrors
332 // ParticleBase::destroy). invalid has length getLocalNum(); invalid[i] sets the
333 // key of owned particle startIndex()+i to removeKey, so the next syncGravBH
334 // drops it and gathers the survivors. Lazy: getLocalNum() is unchanged until
335 // that sync. destroyNum is accepted for ParticleBase parity (unused).
336 void destroy(const bool* invalid, std::size_t /*destroyNum*/) {
337 detail::markRemoved(keys.data() + domain_.startIndex(), invalid,
338 domain_.nParticles(), cstone::removeKey<KeyType>::value);
339 }

452 // After a sync cstone holds valid keys only for the assigned range; the halo slots
453 // keep their previous content, including removeKey markers set by destroy() (sorted
454 // there by the exchange). cstone's computeSfcKeys never overwrites a marker, so a
455 // particle received into such a slot by a later sync would inherit it: it is dropped
456 // from the local set, the assignment counts no longer match and the halo layout
457 // becomes inconsistent ("LET refine"). Reset the halo keys and recompute them from
458 // the halo positions (leaf-based h reads them).
459 void resetHaloKeys() {
460 const LocalIndex start = domain_.startIndex();
461 const LocalIndex end = domain_.endIndex();
462 const LocalIndex n = domain_.nParticlesWithHalos();
463 const auto& bx = domain_.box();
464 cstone::fill(kExec, keys.data(), keys.data() + start, KeyType(0));
465 cstone::fill(kExec, keys.data() + end, keys.data() + n, KeyType(0));
466 cstone::computeSfcKeys(kExec, x.data(), y.data(), z.data(),
467 cstone::sfcKindPointer(keys.data()), start, bx);
468 cstone::computeSfcKeys(kExec, x.data() + end, y.data() + end, z.data() + end,
469 cstone::sfcKindPointer(keys.data()) + end, n - end, bx);
470 }
471
472 // Restore h after syncGravBH (which threads a dummy h=0 through cstone) for
473 // ryoanji P2P softening, over [0, nWithHalos). Uniform fill of newly-grown
474 // entries, or per-particle leaf-edge length when leafBasedH is set.
475 void refillHaloH() {
476 const LocalIndex n = domain_.nParticlesWithHalos();
477 if (leafBasedH) {
478 // Per-particle h = local octree leaf edge length. Recomputed every
479 // step over [0, nWithHalos) because focus-tree refinement migrates
480 // particles between leaves between calls.
481 h.resize(n);
482 auto leaves = domain_.focusTree().treeLeavesAcc();
483 const auto& bx = domain_.box();
484 const Tc vol = bx.lx() * bx.ly() * bx.lz();
485 const Th cbrtVol = static_cast<Th>(std::cbrt(static_cast<double>(vol)));
486 const unsigned maxLevel = cstone::maxTreeLevel<KeyType>{};
487 const auto reductions =
488 cstone::AxesBits{maxLevel, maxLevel, maxLevel} - bx.getBoxDimBits(maxLevel);
489 detail::setHFromLeaves<Th, KeyType>(
490 keys.data(), leaves.data(), static_cast<int>(leaves.size()),
491 cbrtVol, reductions, h.data(), n);
492 } else {
493 // Fill only newly-grown entries — assumes uniform h (the existing
494 // entries already hold the correct value from create()/IC or the
495 // previous grow).
496 const LocalIndex oldH = h.size();
497 h.resize(n);
498 if (n > oldH) {
499 cstone::fill(kExec, h.data() + oldH, h.data() + n, uniformH);
500 }
501 }
502 }
