26// Multipole expansion of the tree cells. Quadrupole (ryoanji::CartesianQuadrupole) has no
27// dipole term: exact when all sources share one sign (the dipole about the |q|-weighted
28// expansion center vanishes), but first-order wrong for cells that mix signs (e.g. image
29// charges). DipoleQuadrupole (ryoanji::CartesianMDQpole) adds the dipole; open boxes only,
30// ryoanji's Ewald summation exists for Quadrupole alone.
31enum class MultipoleOrder { Quadrupole, DipoleQuadrupole };
32
33// Barnes-Hut field solver: build local BH tree, P2M + M2M upsweep, traverse
34// to write Ex/Ey/Ez. When the container's box is fully periodic the canonical
35// Ewald correction (real-space erfc tail + reciprocal-space sum) is appended
36// after the truncated near-field lattice sum; for open BCs Ewald is skipped.
37//
38// Templated on a precision policy P (see NBody/BHPrecision.hpp). The five
39// ryoanji scalar template parameters (Tc, Th, Tm, Ta, Tf) plus the multipole
40// value type (Tmm) all come from P, so a single template parameter selects
41// one of the three pre-compiled ryoanji traversal instantiations.
42template <class P, unsigned Dim>
43class NBodySolver {
44 static_assert(Dim == 3, "NBodySolver requires Dim == 3");
45
46public:
47 using Container = NBodyParticleContainer<P, Dim>;
48 using Tc = typename P::Tc;
49
50 struct Params {
51 Tc G = Tc(1); // Coulomb / gravitational prefactor (position units)
52 float theta = 0.5f; // multipole-acceptance angle
53 int numShells = 1; // BH near-field image-lattice extent (periodic only)
54 ryoanji::EwaldSettings ewaldSettings;
55 MultipoleOrder multipoles = MultipoleOrder::Quadrupole; // fixed per solver instance
56 // Consistency checks of the sources every solve (positions against the box, target-group
57 // extents), printed as [bh.check]. A failed traversal always reports them.
58 bool checks = false;
59 };
60
61 NBodySolver(Container& pc, Params params);
62 ~NBodySolver();
63
64 NBodySolver(const NBodySolver&) = delete;
65 NBodySolver& operator=(const NBodySolver&) = delete;
66 NBodySolver(NBodySolver&&) noexcept;
67 NBodySolver& operator=(NBodySolver&&) noexcept;
68
69 // warmup=true skips the per-scope bh.* IpplTimings and the per-step stats
70 // printout. Used by NBodyManager::pre_run() so the initial t=0 solve does
71 // not pollute the per-step BH timing accumulators (the "firstSolve" outer
72 // timer in pre_run already captures init cost separately).
73 void runSolver(bool warmup = false);
74
75private:
76 class Impl;
77 std::unique_ptr<Impl> impl_;
78};
