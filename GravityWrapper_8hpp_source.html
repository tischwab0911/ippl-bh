18// Local adaptations for the IPPL fork:
19// - namespace ippl::nbody (was sphexa)
20// - scalar types derived via std::declval (our container/DataType is not
21// default-constructible — it owns a cstone::Domain)
22// - setEwaldSettings() so the BH solver can pass numShells / Ewald params

44// ryoanji's Ewald summation exists for CartesianQuadrupole only: the GPU kernels are not
45// instantiated for other types (link error) and the header-only CPU version would silently
46// ignore a dipole. Both multipole types are util::array<T, N> aliases, so is_same works.
47template <class MType>
48inline constexpr bool kHasEwald =
49 std::is_same_v<MType, ryoanji::CartesianQuadrupole<typename MType::value_type>>;
50
51[[noreturn]] inline void throwNoEwald() {
52 throw std::runtime_error(
53 "Barnes-Hut: periodic boxes need Ewald summation, which exists only for quadrupole "
54 "multipoles (MultipoleOrder::Quadrupole).");
55}

58class MultipoleHolderCpu {

68 return {.firstBody = domain.startIndex(),
69 .lastBody = domain.endIndex(),
70 .numGroups = 0,

76 const auto& focusTree = domain.focusTree();
77 reallocate(multipoles_, focusTree.octreeViewAcc().numNodes, 1.05);
78 ryoanji::computeMultipoles(d.x.data(), d.y.data(), d.z.data(), d.m.data(),

83 void traverse(cstone::GroupView /*grp*/, DataType& d, const DomainType& domain) {
84 const auto& focusTree = domain.focusTree();
85 const auto octree = focusTree.octreeViewAcc();
86 const auto& box = domain.box();
87 bool usePbc = box.boundaryX() == cstone::BoundaryType::periodic;
88 int numShells = usePbc ? ewaldSettings_.numReplicaShells : 0;
89
90 d.egrav = 0;
91 ryoanji::computeGravity(octree.childOffsets, octree.parents, octree.internalToLeaf,
92 focusTree.expansionCentersAcc().data(), multipoles_.data(),
93 domain.layout().data(), domain.startCell(), domain.endCell(),
94 d.x.data(), d.y.data(), d.z.data(), d.h.data(), d.m.data(),
95 domain.box(), d.g, d.ugrav.data(), d.ax.data(), d.ay.data(),
96 d.az.data(), &d.egrav, numShells);
97
98 if constexpr (kHasEwald<MType>) {
99 if (usePbc) {
100 ryoanji::computeGravityEwald(makeVec3(focusTree.expansionCentersAcc()[0]),
101 multipoles_.front(), domain.startIndex(),
102 domain.endIndex(), d.x.data(), d.y.data(),
103 d.z.data(), d.m.data(), box, d.g, d.ugrav.data(),
104 d.ax.data(), d.ay.data(), d.az.data(), &d.egrav,
105 ewaldSettings_);
106 }
107 } else if (usePbc) {
108 throwNoEwald();
109 }
110 }
111
112 util::array<uint64_t, 5> readStats() const { return {0, 0, 0, 0, 0}; }
113 uint64_t retriedGroups() const { return 0; }
114
115 const MType* multipoles() const { return multipoles_.data(); }
116
117private:
118 std::vector<MType> multipoles_;
119 ryoanji::EwaldSettings ewaldSettings_;
120};

122template <class MType, class DomainType, class DataType>
123class MultipoleHolderGpu {
124 using KeyType = typename DataType::KeyType;
125 using Tc = typename std::decay_t<decltype(std::declval<DataType>().x)>::value_type;
126 using Th = typename std::decay_t<decltype(std::declval<DataType>().h)>::value_type;
127 using Tm = typename std::decay_t<decltype(std::declval<DataType>().m)>::value_type;
128 using Ta = typename std::decay_t<decltype(std::declval<DataType>().ax)>::value_type;
129 using Tf = typename DomainType::RealType;
130
131public:
132 MultipoleHolderGpu() = default;
133
134 void setEwaldSettings(const ryoanji::EwaldSettings& s) { ewaldSettings_ = s; }
135
136 cstone::GroupView computeSpatialGroups(const DataType& d, const DomainType& domain) {
137 // No h (nullptr): cstone splits groups at min(tolFactor * leaf edge, 2h), an SPH criterion.
138 // With h ~ one particle spacing (BH softening) groups shattered into a few particles, each
139 // costing a warp tile and a tree walk. The leaf term alone keeps groups compact.
140 return mHolder_.computeSpatialGroups(domain.startIndex(), domain.endIndex(), rawPtr(d.x),
141 rawPtr(d.y), rawPtr(d.z),
142 static_cast<const Th*>(nullptr), domain.focusTree(),
143 domain.layout().data(), domain.box());
144 }
145
146 void upsweep(const DataType& d, const DomainType& domain) {
147 const auto& focusTree = domain.focusTree();
148 mHolder_.upsweep(rawPtr(d.x), rawPtr(d.y), rawPtr(d.z), rawPtr(d.m), domain.globalTree(),
149 focusTree, domain.layout().data());
150 }
151
152 void traverse(cstone::GroupView grp, DataType& d, const DomainType& domain) {
153 const auto& box = domain.box();
154 bool usePbc = box.boundaryX() == cstone::BoundaryType::periodic;
155 int numShells = usePbc ? ewaldSettings_.numReplicaShells : 0;
156
157 d.egrav = mHolder_.compute(grp, rawPtr(d.x), rawPtr(d.y), rawPtr(d.z), rawPtr(d.m),
158 rawPtr(d.h), d.g, numShells, domain.box(), rawPtr(d.ugrav),
159 rawPtr(d.ax), rawPtr(d.ay), rawPtr(d.az));
160
161 if constexpr (kHasEwald<MType>) {
162 if (usePbc) {
163 ryoanji::Vec4<Tf> rootCenter;
164 cstone::memcpyD2HAsync(domain.exec(), domain.focusTree().expansionCentersAcc().data(), 1,
165 &rootCenter);
166 MType rootM;
167 cstone::memcpyD2HAsync(domain.exec(), mHolder_.deviceMultipoles(), 1, &rootM);
168 cstone::syncGpu(domain.exec());
169
170 computeGravityEwaldGpu(makeVec3(rootCenter), rootM, grp, rawPtr(d.x), rawPtr(d.y),
171 rawPtr(d.z), rawPtr(d.m), box, d.g, rawPtr(d.ugrav),
172 rawPtr(d.ax), rawPtr(d.ay), rawPtr(d.az), &d.egrav,
173 ewaldSettings_);
174 }
175 } else if (usePbc) {
176 throwNoEwald();
177 }
178 }
179
181 util::array<uint64_t, 5> readStats() const { return mHolder_.readStats(); }
182 uint64_t retriedGroups() const { return mHolder_.retriedGroups(); }
183
184 const MType* deviceMultipoles() const { return mHolder_.deviceMultipoles(); }
185
186private:
187 ryoanji::MultipoleHolder<Tc, Th, Tm, Ta, Tf, KeyType, MType> mHolder_;
188 ryoanji::EwaldSettings ewaldSettings_;
189};
