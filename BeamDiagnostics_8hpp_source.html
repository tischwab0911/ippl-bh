28// CPU (OpenMP) reductions.
29template <class P>
30Triple<typename P::Tc> reduceMeanVelocityCpu(NBodyParticleContainer<P, 3>& pc) {
31 using Tc = typename P::Tc;
32 const long start = static_cast<long>(pc.startIndex());
33 const long end = static_cast<long>(pc.endIndex());
34 Triple<Tc> localSum{Tc(0), Tc(0), Tc(0)};
35 if (end > start) {
36 const Tc* px = getRaw<"Px">(pc); const Tc* py = getRaw<"Py">(pc); const Tc* pz = getRaw<"Pz">(pc);
37 Tc sx = Tc(0), sy = Tc(0), sz = Tc(0);
38#pragma omp parallel for reduction(+ : sx, sy, sz) schedule(static)
39 for (long i = start; i < end; ++i) {
40 sx += px[i];
41 sy += py[i];
42 sz += pz[i];
43 }
44 localSum = {sx, sy, sz};
45 }
46 return finalizeMeanTriple<Tc>(localSum,
47 static_cast<unsigned long>(end > start ? end - start : 0),
48 pc.comm());
49}

51template <class P>
52Triple<typename P::Tc> reduceTemperatureCpu(NBodyParticleContainer<P, 3>& pc,
53 Triple<typename P::Tc> avgV) {
54 using Tc = typename P::Tc;
55 const long start = static_cast<long>(pc.startIndex());
56 const long end = static_cast<long>(pc.endIndex());
57 Triple<Tc> localSum{Tc(0), Tc(0), Tc(0)};
58 if (end > start) {
59 const Tc* px = getRaw<"Px">(pc); const Tc* py = getRaw<"Py">(pc); const Tc* pz = getRaw<"Pz">(pc);
60 const Tc mvx = avgV.x, mvy = avgV.y, mvz = avgV.z;
61 Tc sx = Tc(0), sy = Tc(0), sz = Tc(0);
62#pragma omp parallel for reduction(+ : sx, sy, sz) schedule(static)
63 for (long i = start; i < end; ++i) {
64 const Tc dx = px[i] - mvx;
65 const Tc dy = py[i] - mvy;
66 const Tc dz = pz[i] - mvz;
67 sx += dx * dx;
68 sy += dy * dy;
69 sz += dz * dz;
70 }
71 localSum = {sx, sy, sz};
72 }
73 return finalizeMeanTriple<Tc>(localSum,
74 static_cast<unsigned long>(end > start ? end - start : 0),
75 pc.comm());
76}

80// Mean velocity over [startIndex(), endIndex()), reading container-resident
81// Px/Py/Pz. Dispatches to the thrust (GPU) or OpenMP (CPU) body at compile time.
82template <class P>
83Triple<typename P::Tc> reduceMeanVelocity(NBodyParticleContainer<P, 3>& pc) {
84 using Tc = typename P::Tc;
85 if constexpr (kHaveGpu) {
86 const unsigned start = pc.startIndex();
87 const unsigned end = pc.endIndex();
88 const std::size_t n = (end > start) ? end - start : 0u;
89 Triple<Tc> localSum{Tc(0), Tc(0), Tc(0)};
90 if (n > 0) {
91 localSum = detail::sumTripleGpu<Tc>(getRaw<"Px">(pc) + start,
92 getRaw<"Py">(pc) + start,
93 getRaw<"Pz">(pc) + start, n);
94 }
95 return detail::finalizeMeanTriple<Tc>(localSum, static_cast<unsigned long>(n), pc.comm());
96 } else {
97 return detail::reduceMeanVelocityCpu(pc);
98 }
99}

101// Per-axis variance of velocity around (avgVx, avgVy, avgVz).
102template <class P>
103Triple<typename P::Tc> reduceTemperature(NBodyParticleContainer<P, 3>& pc,
104 Triple<typename P::Tc> avgV) {
105 using Tc = typename P::Tc;
106 if constexpr (kHaveGpu) {
107 const unsigned start = pc.startIndex();
108 const unsigned end = pc.endIndex();
109 const std::size_t n = (end > start) ? end - start : 0u;
110 Triple<Tc> localSum{Tc(0), Tc(0), Tc(0)};
111 if (n > 0) {
112 localSum = detail::varianceTripleGpu<Tc>(getRaw<"Px">(pc) + start,
113 getRaw<"Py">(pc) + start,
114 getRaw<"Pz">(pc) + start, n,
115 avgV.x, avgV.y, avgV.z);
116 }
117 return detail::finalizeMeanTriple<Tc>(localSum, static_cast<unsigned long>(n), pc.comm());
118 } else {
119 return detail::reduceTemperatureCpu(pc, avgV);
120 }
121}
